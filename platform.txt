# XR871 Arduino Platform Configuration v1.3.0
# Optimized for Arduino IDE build system

# ============================================================
# Compiler configuration
# ============================================================
compiler.path={runtime.tools.arm-none-eabi-gcc.path}/bin/

compiler.c.cmd=arm-none-eabi-gcc
compiler.cpp.cmd=arm-none-eabi-g++
compiler.as.cmd=arm-none-eabi-as
compiler.ar.cmd=arm-none-eabi-ar
compiler.ld.cmd=arm-none-eabi-ld
compiler.objcopy.cmd=arm-none-eabi-objcopy
compiler.objdump.cmd=arm-none-eabi-objdump
compiler.size.cmd=arm-none-eabi-size
compiler.strip.cmd=arm-none-eabi-strip
compiler.readelf.cmd=arm-none-eabi-readelf
compiler.nm.cmd=arm-none-eabi-nm

# Cortex-M4F CPU flags
compiler.cpu.flags=-mcpu=cortex-m4 -mthumb -mfpu=fpv4-sp-d16 -mfloat-abi=softfp

# Common C flags (with SDK warning suppression for GCC 13)
compiler.c.flags=-c -Os -w -std=gnu99 -ffunction-sections -fdata-sections
compiler.c.flags+= -nostdlib -nostdinc
compiler.c.flags+= {compiler.cpu.flags}
compiler.c.flags+= -D__CONFIG_CHIP_XR871 -D__CONFIG_CPU_CM4F -D__CONFIG_OS_FREERTOS
compiler.c.flags+= -D__CONFIG_LIBC_REDEFINE_GCC_INT32_TYPE
compiler.c.flags+= -D__CONFIG_LIBC_PRINTF_FLOAT -D__CONFIG_LIBC_WRAP_STDIO
compiler.c.flags+= -DARDUINO=10819 -DARDUINO_XR871 -DARDUINO_ARCH_XR871
compiler.c.flags+= -I{runtime.platform.path}/SDK/include
compiler.c.flags+= -I{runtime.platform.path}/SDK/include/driver
compiler.c.flags+= -I{runtime.platform.path}/SDK/include/driver/chip
compiler.c.flags+= -I{runtime.platform.path}/SDK/include/driver/cmsis
compiler.c.flags+= -I{runtime.platform.path}/SDK/include/kernel
compiler.c.flags+= -I{runtime.platform.path}/SDK/include/kernel/os
compiler.c.flags+= -I{runtime.platform.path}/SDK/include/kernel/os/FreeRTOS
compiler.c.flags+= -I{runtime.platform.path}/SDK/include/kernel/FreeRTOS
compiler.c.flags+= -I{runtime.platform.path}/SDK/include/kernel/FreeRTOS/portable/GCC/ARM_CM4F
compiler.c.flags+= -I{runtime.platform.path}/SDK/include/libc
compiler.c.flags+= -I{runtime.platform.path}/SDK/project/common/board
compiler.c.flags+= -I{runtime.platform.path}/SDK/project/common/board/xr871_evb_main
compiler.c.flags+= -I{runtime.platform.path}/variants/{build.variant}

# C++ flags
compiler.cpp.flags={compiler.c.flags} -fno-rtti -fno-exceptions -std=gnu++98

# Assembler flags
compiler.as.flags=-c -x assembler-with-cpp {compiler.cpu.flags}

# Linker flags
compiler.ld.flags=-nostartfiles -nostdlib -Wl,--gc-sections
compiler.ld.flags+= {compiler.cpu.flags}
compiler.ld.flags+= -T{runtime.platform.path}/SDK/project/linker_script/gcc/xr871/appos.ld
compiler.ld.flags+= -L{runtime.platform.path}/lib
compiler.ld.flags+= -L{runtime.platform.path}/SDK/lib
compiler.ld.flags+= -Wl,--whole-archive -lxr871 -Wl,--no-whole-archive
compiler.ld.flags+= -lc -lgcc -lnosys

# Archive flags
compiler.ar.flags=rcs

# ============================================================
# Build recipes (Arduino IDE build system)
# ============================================================

# C file compilation
recipe.c.o.pattern="{compiler.path}{compiler.c.cmd}" {compiler.c.flags} {source_file} -o {object_file}

# C++ file compilation
recipe.cpp.o.pattern="{compiler.path}{compiler.cpp.cmd}" {compiler.cpp.flags} {source_file} -o {object_file}

# Assembly compilation
recipe.S.o.pattern="{compiler.path}{compiler.c.cmd}" -x assembler-with-cpp {compiler.as.flags} {source_file} -o {object_file}

# Link object files into ELF
recipe.c.combine.pattern="{compiler.path}{compiler.ld.cmd}" {compiler.ld.flags} -Wl,-Map={build.path}/{build.project_name}.map -o "{build.path}/{build.project_name}.elf" {object_files}

# Generate binary from ELF
recipe.objcopy.bin.pattern="{compiler.path}{compiler.objcopy.cmd}" -O binary "{build.path}/{build.project_name}.elf" "{build.path}/{build.project_name}.bin"

# Size report
recipe.size.pattern="{compiler.path}{compiler.size.cmd}" -B "{build.path}/{build.project_name}.elf"
recipe.size.regex=^(?:\.text|\.data|\.rodata|\.boot2|\.vector_table)\s+([0-9]+).*
recipe.data.regex=^(?:\.data|\.bss|\.noinit)\s+([0-9]+).*

# ============================================================
# Upload configuration (Python BROM prioritized, phoenixMC fallback)
# ============================================================
tools.xr871_tool.path={runtime.tools.xr871_tool.path}
tools.xr871_tool.cmd=python3 {runtime.tools.xr871_tool.path}/scripts/flasher.py

tools.xr871_tool.upload.protocol=serial
tools.xr871_tool.upload.params.verbose=-v
tools.xr871_tool.upload.params.quiet=-q

# Upload pattern: auto-select BROM or phoenixMC
tools.xr871_tool.upload.pattern="{cmd}" "{serial.port}" {upload.speed} "{build.path}/{build.project_name}.bin" --method auto

# Erase pattern
tools.xr871_tool.erase.pattern="{cmd}" "{serial.port}" {upload.speed} --erase

# ============================================================
# Network upload (OTA)
# ============================================================
tools.xr871_tool.network_cmd=python3 {runtime.tools.xr871_tool.path}/scripts/flasher.py
tools.xr871_tool.upload.network_pattern="{network_cmd}" "{serial.port}" 8266 "{build.path}/{build.project_name}.bin" --method auto
